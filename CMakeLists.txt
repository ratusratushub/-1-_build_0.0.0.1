cmake_minimum_required(VERSION 3.24)
project(SpaceFoundation VERSION 0.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SPACE_BUILD_APP "Build native OpenGL application and GPU tests" ON)
include(CTest)
find_package(Threads REQUIRED)
add_library(space_cpu src/SpatialGrid.cpp)
target_include_directories(space_cpu PUBLIC include)
target_link_libraries(space_cpu PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(space_cpu PRIVATE /W4)
else()
  target_compile_options(space_cpu PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(space_cpu_tests tests/cpu_tests.cpp)
  target_link_libraries(space_cpu_tests PRIVATE space_cpu)
  add_test(NAME cpu_foundation COMMAND space_cpu_tests)
endif()
if(SPACE_BUILD_APP)
  include(FetchContent)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
  FetchContent_Declare(glad GIT_REPOSITORY https://github.com/Dav1dde/glad.git GIT_TAG v2.0.8 GIT_SHALLOW TRUE SOURCE_SUBDIR cmake)
  FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1 GIT_SHALLOW TRUE)
  FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.8 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(glfw glad glm imgui)
  glad_add_library(space_gl REPRODUCIBLE API gl:core=4.3)
  add_library(space_imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(space_imgui PUBLIC ${imgui_SOURCE_DIR})
  target_compile_definitions(space_imgui PRIVATE GLFW_INCLUDE_NONE)
  target_link_libraries(space_imgui PUBLIC glfw ${CMAKE_DL_LIBS})
  add_library(space_core src/EntityManager.cpp src/ComputePipeline.cpp src/GLUtils.cpp)
  target_link_libraries(space_core PUBLIC space_cpu space_gl)
  if(MSVC)
    target_compile_options(space_core PRIVATE /W4)
  else()
    target_compile_options(space_core PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_executable(space_sim src/main.cpp src/Renderer.cpp src/Overlay.cpp)
  target_link_libraries(space_sim PRIVATE space_core glfw glm::glm space_imgui)
  target_compile_definitions(space_sim PRIVATE SPACE_SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/shaders")
  if(BUILD_TESTING)
    add_executable(space_gpu_tests tests/gpu_tests.cpp)
    target_link_libraries(space_gpu_tests PRIVATE space_core glfw)
    target_compile_definitions(space_gpu_tests PRIVATE SPACE_SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/shaders")
    add_test(NAME gpu_reference COMMAND space_gpu_tests)
    set_tests_properties(gpu_reference PROPERTIES SKIP_RETURN_CODE 77 LABELS "gpu")
  endif()
endif()
